  // --- GLOBAL CLIENT-SIDE IMAGE COMPRESSOR ---
  function compressImage(file, callback) {
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function(event) {
      const img = new Image();
      img.onload = function() {
        const canvas = document.createElement('canvas');
        const MAX_WIDTH = 1000;
        const MAX_HEIGHT = 1000;
        let width = img.width;
        let height = img.height;
        
        if (width > height && width > MAX_WIDTH) {
          height *= MAX_WIDTH / width; width = MAX_WIDTH;
        } else if (height > MAX_HEIGHT) {
          width *= MAX_HEIGHT / height; height = MAX_HEIGHT;
        }
        
        canvas.width = width; canvas.height = height;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, width, height);
        
        // Export as 60% quality JPEG for maximum speed
        callback(canvas.toDataURL('image/jpeg', 0.6)); 
      };
      img.src = event.target.result;
    };
    reader.readAsDataURL(file);
  }

  function smartValidateTime(masterId, actionType, inputTime) {
    if (!inputTime) return true; // Let standard empty-field validations handle this
    
    // 1. Admin/Back-door Bypass: If overriding the date, allow free time travel
    if (typeof overrideSystemDate !== 'undefined' && overrideSystemDate) return true; 

    // 2. Locate the specific Master Batch and its latest segment
    let master = globalUnifiedData.masterCards.find(m => m.masterId === masterId);
    if (!master || !master.history || master.history.length === 0) return true;

    let lastSeg = master.history[master.history.length - 1];
    let todayStr = getActiveDateStr(); 
    let inputMins = getMins(inputTime);

    // 3. Chronological Logic Verification
    if (actionType === 'pause' || actionType === 'end') {
        // Only restrict if the session STARTED today
        if (lastSeg.status === 'Active' && lastSeg.date === todayStr) {
            let startMins = getMins(lastSeg.startTime);
            if (inputMins < startMins) {
                showSmartAlert(
                  "تنبيه: تسلسل زمني غير منطقي", 
                  `لا يمكن إنهاء التشغيلة في الساعة ${inputTime} لأنها بدأت في الساعة ${lastSeg.startTime} من نفس اليوم.\n\nهل قمت بإدخال (ص/م) بشكل خاطئ؟`
                );
                return false;
            }
        }
    } else if (actionType === 'resume') {
        // Only restrict if the session was PAUSED today
        if (lastSeg.status === 'Paused' && lastSeg.date === todayStr) {
            let pausedMins = getMins(lastSeg.endTime);
            if (inputMins < pausedMins) {
                showSmartAlert(
                  "تنبيه: تسلسل زمني غير منطقي", 
                  `لا يمكن استئناف العمل في الساعة ${inputTime} لأن التشغيلة أُوقفت في الساعة ${lastSeg.endTime} من نفس اليوم.\n\nهل قمت بإدخال (ص/م) بشكل خاطئ؟`
                );
                return false;
            }
        }
    }
    
    // If it spans across midnight (different days), or passes the math above, it is Valid.
    return true;
  }

  function formatDuration(mins) {
    if (!mins) return '0 د';
    let h = Math.floor(mins / 60);
    let m = mins % 60;
    if (h > 0 && m > 0) return h + ' س و ' + m + ' د';
    if (h > 0) return h + ' س';
    return m + ' د';
  }

  function getMins(t) {
    if(!t) return 0;
    let p = String(t).split(':');
    return parseInt(p[0]||0)*60 + parseInt(p[1]||0);
  }
  
  function getHM(m) {
    let h = Math.floor(m/60); let mn = m%60;
    return (h<10?'0':'')+h+':'+(mn<10?'0':'')+mn;
  }

function submitProdAction(action) {
    const eTime = document.getElementById('prodEndTime').value;
    const qty = document.getElementById('prodEndQty').value;
    const notes = document.getElementById('prodEndNotes').value;
    const masterSessionId = document.getElementById('endSessionId').value;

    if (!eTime) return alert("يرجى إدخال الوقت المحدد");
    if (action === 'complete' && !qty) return alert("يرجى إدخال الكمية المنتجة قبل الإنهاء");
    if ((action === 'pause' || action === 'cancel') && !notes) return alert("يرجى ذكر السبب في حقل الملاحظات");
    
    // --- SMART TIME VALIDATOR HOOK ---
    if (!smartValidateTime(masterSessionId, action, eTime)) return;
    // ---------------------------------
    
    document.getElementById('endModal').style.display = 'none';
    showLoading();
    
    const data = {
      sessionId: document.getElementById('endSessionId').value,
      user: currentUser, endTime: eTime, quantity: parseFloat(qty) || 0,
      notes: notes, action: action
    };
    
    if (uploadedProdImage) {
      google.script.run
        .withSuccessHandler(function(res) {
          data.imageUrl = res.url;
          finalizeEndProd(data); // Safely utilizes your existing finalize hook
        })
        .withFailureHandler(function(e) { hideLoading(); alert("خطأ في رفع الصورة: " + e.message); finalizeEndProd(data); })
        .uploadImageToDrive(uploadedProdImage, 'Prod_' + data.sessionId + '.jpg');
    } else {
      finalizeEndProd(data);
    }
  }

  function openEditModal(id) {
    document.getElementById('editSessionId').value = id; // Saves the clicked ID
    
    let isMaster = false; let isSegment = false; let isTimelineOnly = false;
    let masterObj = globalUnifiedData.masterCards.find(m => m.masterId === id);
    let segmentObj = null;
    
    // Determine the Data Type
    if (masterObj) {
       isMaster = true;
    } else {
       for (let m of globalUnifiedData.masterCards) {
          let foundSeg = m.history.find(s => s.segmentId === id);
          if (foundSeg) { segmentObj = foundSeg; masterObj = m; isSegment = true; break; }
       }
    }
    
    let timelineObj = globalUnifiedData.timeline.find(t => t.segmentId === id || t.masterId === id);
    if (!isMaster && !isSegment && timelineObj) isTimelineOnly = true;

    if (!isMaster && !isSegment && !isTimelineOnly) return showSmartAlert("تنبيه", "عذراً، لم يتم العثور على البيانات في الذاكرة.");

    // Configure Modal UI Based on Type
    if (isTimelineOnly) {
       if (timelineObj.product === 'وقت الراحة') {
          document.getElementById('editCompletedFields').style.display = 'none';
          document.getElementById('editTimeFields').style.display = 'block';
          document.getElementById('editStartTime').value = timelineObj.startTime;
          document.getElementById('editEndTimeGroup').style.display = (timelineObj.status === 'Active') ? 'none' : 'block';
          document.getElementById('editEndTime').value = timelineObj.endTime || '';
       } else if (timelineObj.product === 'وقت مهدر') {
          document.getElementById('editTimeFields').style.display = 'none';
          document.getElementById('editCompletedFields').style.display = 'block';
          document.getElementById('editQtyGroup').style.display = 'none';
       }
    } 
    else if (isSegment) {
       document.getElementById('editTimeFields').style.display = 'block';
       document.getElementById('editCompletedFields').style.display = 'block'; // Unlock Image/Notes for Segments
       if (document.getElementById('editQtyGroup')) document.getElementById('editQtyGroup').style.display = 'none'; // Hide Qty for Segments
       document.getElementById('editStartTime').value = segmentObj.startTime;
       document.getElementById('editEndTimeGroup').style.display = (segmentObj.status === 'Active') ? 'none' : 'block';
       document.getElementById('editEndTime').value = segmentObj.endTime || '';
       document.getElementById('editNotes').value = segmentObj.notes || '';
    } 
    else if (isMaster) {
       document.getElementById('editTimeFields').style.display = 'none'; // Master edits don't touch specific time
       document.getElementById('editCompletedFields').style.display = 'block';
       if (document.getElementById('editQtyGroup')) document.getElementById('editQtyGroup').style.display = 'block';
       document.getElementById('editTotalQty').value = parseFloat(masterObj.totalQuantity.toFixed(1)) || 0;
       if (document.getElementById('editSegmentQty')) document.getElementById('editSegmentQty').textContent = masterObj.totalQuantity || 0;
       let lastSeg = masterObj.history[masterObj.history.length - 1];
       document.getElementById('editNotes').value = lastSeg.notes || '';
    }

    uploadedEditImage = null;
    document.getElementById('editPreviewContainer').style.display = 'none';
    document.getElementById('btnUploadEditImg').style.display = 'flex';
    document.getElementById('editImageInput').value = '';
    document.getElementById('editModal').style.display = 'flex';
  }

  function setRepQuickDate(type) {
    const today = new Date();
    const startInput = document.getElementById('repStartDate');
    const endInput = document.getElementById('repEndDate');
    
    if (type === 'all') {
        startInput.value = '';
        endInput.value = '';
        document.getElementById('repProductTable').innerHTML = '<tr><td colspan="6" class="px-6 py-10 text-center text-slate-400"><i class="fas fa-spinner fa-spin text-2xl mb-2"></i><br>جاري جلب السجلات...</td></tr>';
        
        google.script.run.withSuccessHandler(function(res) {
            if (res.min) startInput.value = res.min;
            if (res.max) endInput.value = res.max;
            loadExecutiveReport();
        }).getProductionDateRange();
        return;
    }
    
    let start = new Date();
    let end = new Date();
    
    if (type === 'today') {
        // Keep as today
    } else if (type === 'week') {
        let day = today.getDay(); // 0 is Sunday, 6 is Saturday
        let diff = day === 6 ? 0 : day + 1; // Subtract to reach last Saturday
        start.setDate(today.getDate() - diff);
    } else if (type === 'month') {
        start.setDate(1);
        end = new Date(today.getFullYear(), today.getMonth() + 1, 0); // Last day
    }
    
    // Format YYYY-MM-DD safely
    const yS = start.getFullYear(), mS = String(start.getMonth() + 1).padStart(2, '0'), dS = String(start.getDate()).padStart(2, '0');
    const yE = end.getFullYear(), mE = String(end.getMonth() + 1).padStart(2, '0'), dE = String(end.getDate()).padStart(2, '0');
    
    startInput.value = `${yS}-${mS}-${dS}`;
    endInput.value = `${yE}-${mE}-${dE}`;
    
    loadExecutiveReport();
  }

  function filterTable() {
    const searchInput = document.getElementById('searchInput');
    const searchTerm = searchInput.value.trim().toLowerCase();
    const clearBtn = document.getElementById('clearSearchBtn');
    
    clearBtn.style.display = searchTerm ? 'block' : 'none';
    
    if (!searchTerm) {
      filteredReportData = fullReportData;
      displayReport(fullReportData);
      return;
    }
    
    filteredReportData = fullReportData.filter(function(row) {
      const rowText = Object.values(row).join('|').toLowerCase();
      return rowText.includes(searchTerm);
    });
    
    displayReport(filteredReportData);
    highlightSearchTerm(searchTerm);
  }

  function highlightSearchTerm(term) {
    const tbody = document.getElementById('reportTableBody');
    const cells = tbody.getElementsByTagName('td');
    
    Array.from(cells).forEach(function(cell) {
      const text = cell.textContent;
      const lowerText = text.toLowerCase();
      const index = lowerText.indexOf(term);
      
      if (index !== -1) {
        const before = text.substring(0, index);
        const match = text.substring(index, index + term.length);
        const after = text.substring(index + term.length);
        
        cell.innerHTML = before + '<span class="highlight">' + match + '</span>' + after;
      }
    });
  }

  function formatDurationHHMMSS(mins) {
    if (!mins || isNaN(mins) || mins === Infinity) return '00:00:00';
    let totalSeconds = Math.round(mins * 60);
    let h = Math.floor(totalSeconds / 3600);
    let m = Math.floor((totalSeconds % 3600) / 60);
    let s = totalSeconds % 60;
    return (h < 10 ? '0' : '') + h + ':' + (m < 10 ? '0' : '') + m + ':' + (s < 10 ? '0' : '') + s;
  }

  function formatReportDuration(mins) {
    if (!mins || isNaN(mins)) return '00:00:00';
    const totalSeconds = Math.round(mins * 60);
    const h = Math.floor(totalSeconds / 3600);
    const m = Math.floor((totalSeconds % 3600) / 60);
    const s = totalSeconds % 60;
    return String(h).padStart(2, '0') + ':' + String(m).padStart(2, '0') + ':' + String(s).padStart(2, '0');
  }

  function renderExecutiveReport(data) {
    hideLoading();
    // Populate Dropdowns on first load

      const pSelect = document.getElementById('repProduct');
      data.productNames.forEach(p => { pSelect.add(new Option(p, p)); });
      filtersLoaded = true;
    }

    // Store globally for Modal breakdown interaction
    window.currentProductStats = data.products;

    // Update KPIs
    document.getElementById('repKpiProd').textContent = formatReportDuration(data.kpis.prod);
    document.getElementById('repKpiWaste').textContent = formatReportDuration(data.kpis.waste);
    document.getElementById('repKpiBreak').textContent = formatReportDuration(data.kpis.break);
    document.getElementById('repKpiOt').textContent = formatReportDuration(data.kpis.overtime);
    
    let effTotal = data.kpis.prod + data.kpis.waste;
    let oee = effTotal > 0 ? ((data.kpis.prod / effTotal) * 100).toFixed(1) : 0;
    document.getElementById('repKpiOee').textContent = oee + '%';

    // Render Modern Interactive Table
    let tbHtml = '';
    if (data.products.length === 0) {
       tbHtml = '<tr><td colspan="6" class="px-6 py-10 text-center text-slate-400">لا توجد بيانات لهذه الفترة</td></tr>';
    } else {
       let totalFactoryTime = data.kpis.prod;

       data.products.forEach((p, index) => {
         let aggCostPct = totalFactoryTime > 0 ? (p.dur / totalFactoryTime) * 100 : 0;
         let loadedTimePerUnit = p.qty > 0 ? (p.dur / p.qty) : 0;
         let unitWeightPct = p.qty > 0 ? (aggCostPct / p.qty) : 0;
         
         tbHtml += `<tr class="hover:bg-slate-50 border-b border-slate-100 transition-colors cursor-pointer group" onclick="openProductDetailsModal(${index})">
            <td class="px-4 py-3 font-bold text-slate-700 border-l border-slate-100 text-right">
               <i class="fas fa-search-plus text-slate-300 group-hover:text-blue-500 transition-colors ml-2" style="font-size: 0.8rem;"></i>
               ${p.name}
            </td>
            <td class="px-4 py-3 text-center border-l border-slate-100">
               <span class="bg-slate-100 text-slate-800 px-3 py-1 rounded-full text-xs font-bold shadow-sm">${p.qty}</span>
            </td>
            <td class="px-4 py-3 text-center border-l border-slate-100 text-slate-600 font-medium tracking-wider" dir="ltr">
               ${formatDurationHHMMSS(p.dur)}
            </td>
            <td class="px-4 py-3 text-center border-l border-slate-100 bg-blue-50 text-blue-700 font-bold tracking-wider">
               <span class="bg-blue-100 px-2 py-1 rounded shadow-sm">${aggCostPct.toFixed(1)}%</span>
            </td>
            <td class="px-4 py-3 text-center border-l border-slate-100 bg-green-50 text-green-700 font-bold tracking-wider" dir="ltr">
               ${formatDurationHHMMSS(loadedTimePerUnit)}
            </td>
            <td class="px-4 py-3 text-center bg-purple-50 text-purple-700 font-bold tracking-wider">
               ${unitWeightPct.toFixed(3)}%
            </td>
         </tr>`;
       });
    }
    document.getElementById('repProductTable').innerHTML = tbHtml;

    // Destroy old charts to prevent overlapping
    if (repDoughnutInstance) repDoughnutInstance.destroy();

    // Draw Doughnut Chart
    const ctxD = document.getElementById('repDoughnutChart').getContext('2d');
    repDoughnutInstance = new Chart(ctxD, {
      type: 'doughnut',
      data: {
        labels: ['الإنتاج', 'المهدر', 'الراحة'],
        datasets: [{
          data: [data.kpis.prod, data.kpis.waste, data.kpis.break],
          backgroundColor: ['#10b981', '#ef4444', '#fbbf24'],
          borderWidth: 0,
          hoverOffset: 4
        }]
      },
      options: {
        responsive: true, maintainAspectRatio: false,
        plugins: { legend: { position: 'bottom', labels: { font: { family: 'inherit', weight: 'bold' } } } }
      }
    });

    // Generate Fully Absorbed Cost Allocation Grid (Replaces Trend Chart)
    const totalPaidTime = data.kpis.prod + data.kpis.waste + data.kpis.break;
    const totalProdTime = data.kpis.prod;

function loadTDSDetails() {
  const product = document.getElementById('tdsProductSelect').value;
  if(!product) return;
  
  showLoading();
  google.script.run
    .withSuccessHandler(function(details) {
      hideLoading();
      tdsDetails = details;
      
      // Auto-calculate Expiry if Date is set
      calculateExpiry();
    })
    .withFailureHandler(error => { hideLoading(); alert(error.message); })
    .getTDSDetails(product);
}

function calculateExpiry() {
  const prodDateStr = document.getElementById('tdsDate').value;
  if (!prodDateStr || !tdsDetails.shelfLife) return;
  
  // Extract number from "24 Months" string
  const months = parseInt(tdsDetails.shelfLife);
  if(isNaN(months)) return;
  
  const prodDate = new Date(prodDateStr);
  const expDate = new Date(prodDate);
  expDate.setMonth(expDate.getMonth() + months);
  // Subtract 1 day for proper expiry
  expDate.setDate(expDate.getDate() - 1);
  
  document.getElementById('tdsExpDate').valueAsDate = expDate;
}

function filterMntDashboard() {
  const taskType = document.getElementById('mntFilterTaskType').value;
  const location = document.getElementById('mntFilterLocation').value;
  const category = document.getElementById('mntFilterCategory').value;
  const status = document.getElementById('mntFilterStatus').value;
  const search = (document.getElementById('mntSearchInput').value || '').trim().toLowerCase();

  const grid = document.getElementById('mntItemsGrid');
  grid.innerHTML = '';

  const filtered = mntMasterData.items.filter(item => {
    if (location !== 'الكل' && item.location !== location) return false;
    if (category !== 'الكل' && item.category !== category) return false;
    
    if (search) {
      const matchId = item.id.toLowerCase().includes(search);
      const matchName = item.name.toLowerCase().includes(search);
      if (!matchId && !matchName) return false;
    }

    if (status !== 'الكل') {
      const cleanMatch = item.cleaning.status === status;
      const maintMatch = item.maintenance.status === status;
      if (taskType === 'نظافة' && !cleanMatch) return false;
      if (taskType === 'صيانة' && !maintMatch) return false;
      if (taskType === 'الكل' && !cleanMatch && !maintMatch) return false;
    }

    return true;
  });

  if (filtered.length === 0) {
    grid.innerHTML = '<div class="col-span-full py-12 text-center text-slate-400 font-bold">لا توجد بنود مطابقة للفلاتر المحددة</div>';
    return;
  }

  filtered.forEach(item => {
    const showClean = (taskType === 'الكل' || taskType === 'نظافة') && item.cleanDays !== null;
    const showMaint = (taskType === 'الكل' || taskType === 'صيانة') && item.maintDays !== null;

    const card = document.createElement('div');
    card.className = 'bg-white rounded-2xl p-5 border border-slate-200/80 shadow-sm flex flex-col justify-between hover:shadow-md transition-all';

    let cleanBadge = '';
    if (showClean) {
      let badgeColor = item.cleaning.status === 'overdue' ? 'bg-rose-50 text-rose-700 border-rose-200' : (item.cleaning.status === 'due_today' ? 'bg-amber-50 text-amber-700 border-amber-200' : 'bg-emerald-50 text-emerald-700 border-emerald-200');
      let badgeText = item.cleaning.status === 'overdue' ? `متأخر (${item.cleaning.daysLeft !== null ? Math.abs(item.cleaning.daysLeft) + ' يوم' : 'جديد'})` : (item.cleaning.status === 'due_today' ? 'مستحق اليوم' : `ساري (${item.cleaning.daysLeft} يوم)`);
      cleanBadge = `
        <div class="p-3 bg-slate-50 rounded-xl border border-slate-100 flex flex-col gap-1.5 mb-2">
          <div class="flex justify-between items-center">
            <span class="text-xs font-bold text-slate-600"><i class="fas fa-broom text-blue-500 ml-1"></i> نظافة (كل ${item.cleanDays} يوم)</span>
            <span class="text-[11px] font-black px-2 py-0.5 rounded-full border ${badgeColor}">${badgeText}</span>
          </div>
          <div class="flex justify-between text-[11px] text-slate-500 font-medium">
            <span>آخر إنجاز: ${item.cleaning.lastDate}</span>
            <span>القادم: ${item.cleaning.nextDate}</span>
          </div>
          <button onclick="quickLogMnt('${item.id}', 'نظافة')" class="mt-1 w-full py-1.5 bg-blue-50 hover:bg-blue-100 text-blue-700 font-bold rounded-lg text-xs transition-colors border border-blue-200 flex items-center justify-center gap-1.5">
            <i class="fas fa-check"></i> توثيق النظافة
          </button>
        </div>
      `;
    }

    let maintBadge = '';
    if (showMaint) {
      let badgeColor = item.maintenance.status === 'overdue' ? 'bg-rose-50 text-rose-700 border-rose-200' : (item.maintenance.status === 'due_today' ? 'bg-amber-50 text-amber-700 border-amber-200' : 'bg-emerald-50 text-emerald-700 border-emerald-200');
      let badgeText = item.maintenance.status === 'overdue' ? `متأخر (${item.maintenance.daysLeft !== null ? Math.abs(item.maintenance.daysLeft) + ' يوم' : 'جديد'})` : (item.maintenance.status === 'due_today' ? 'مستحق اليوم' : `ساري (${item.maintenance.daysLeft} يوم)`);
      maintBadge = `
        <div class="p-3 bg-slate-50 rounded-xl border border-slate-100 flex flex-col gap-1.5">
          <div class="flex justify-between items-center">
            <span class="text-xs font-bold text-slate-600"><i class="fas fa-wrench text-amber-500 ml-1"></i> صيانة (كل ${item.maintDays} يوم)</span>
            <span class="text-[11px] font-black px-2 py-0.5 rounded-full border ${badgeColor}">${badgeText}</span>
          </div>

function handleMntMultiFiles(e) {
  const files = Array.from(e.target.files);
  if (!files || files.length === 0) return;

  showLoading();
  let loaded = 0;

  files.forEach(file => {
    compressImage(file, function(compressedData) {
      uploadedMntImages.push(compressedData);
      loaded++;
      if (loaded === files.length) {
        hideLoading();
        renderMntImageTray();
      }
    });
  });

  e.target.value = ''; // Reset file input
}

function renderMntImageTray() {

}

function submitMntLog(e) {
  e.preventDefault();

  const itemId = document.getElementById('mntFormItem').value;
  const date = document.getElementById('mntTaskDate').value;
  const notes = document.getElementById('mntNotesInput').value.trim();

  if (!itemId) return alert('الرجاء اختيار البند');
  if (!date) return alert('الرجاء تحديد تاريخ الإنجاز');

  // Collect Multi-Performers and validate mandatory 1-10 Ratings for cleaning
  const performerRows = document.querySelectorAll('.mnt-performer-row');
  const performers = [];
  let missingRatingPerformer = null;
  let missingNameCount = 0;

  performerRows.forEach(row => {
    const nameInput = row.querySelector('.mnt-perf-name');
    const ratingInput = row.querySelector('.mnt-perf-rating');
    const name = nameInput ? nameInput.value.trim() : '';
    let rating = null;

    if (!name) {
      missingNameCount++;
      return;
    }

    if (activeMntTaskType === 'نظافة') {
      const rawRating = ratingInput ? convertArabicToEnglishNums(ratingInput.value).trim() : '';
      if (!rawRating) {
        missingRatingPerformer = name;
      } else {
        const parsedRating = parseInt(rawRating);
        if (isNaN(parsedRating) || parsedRating < 1 || parsedRating > 10) {
          missingRatingPerformer = name;
        } else {
          rating = parsedRating;
        }
      }
    }

    performers.push({ name: name, rating: rating });
  });

  if (missingNameCount > 0 || performers.length === 0) {
    return alert('الرجاء إدخال اسم القائم بالمهمة لجميع الحقول');
  }

  if (activeMntTaskType === 'نظافة' && missingRatingPerformer !== null) {
    return alert(`الرجاء إدخال تقييم النظافة (درجة من 1 إلى 10) للعامل: "${missingRatingPerformer}" (حقل إجباري)`);
  }

  const item = mntMasterData.items.find(i => i.id === itemId);

  const payload = {
    taskType: activeMntTaskType,
    itemId: itemId,
    itemName: item ? item.name : '',
    category: item ? item.category : '',
    location: item ? item.location : '',
    date: date,
    performers: performers,
    recorder: currentUser,
    notes: notes,
    images: uploadedMntImages
  };

  showLoading();
  google.script.run
    .withSuccessHandler(function(res) {
      hideLoading();
      document.getElementById('mntLogSuccessMsg').style.display = 'flex';
      document.getElementById('mntLogForm').reset();
      uploadedMntImages = [];
      renderMntImageTray();
      document.getElementById('mntProcedureGuide').style.display = 'none';

      setTimeout(() => {
        document.getElementById('mntLogSuccessMsg').style.display = 'none';
        initPortalMNT(); // Refresh state & return to dashboard
        switchMntTab('dashboard');
      }, 1500);
    })
    .withFailureHandler(function(err) {
      hideLoading();
      alert('خطأ في حفظ البيانات: ' + err.message);
    })
    .saveMntLog(payload);
}

function setMntHistQuickDate(type) {
  const today = new Date();
  const startInput = document.getElementById('mntHistStartDate');
  const endInput = document.getElementById('mntHistEndDate');
  const formatLocal = (d) => `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;

  if (type === 'today') {
    startInput.value = formatLocal(today);
    endInput.value = formatLocal(today);
  } else if (type === 'week') {
    const day = today.getDay();
    const diff = day === 6 ? 0 : day + 1;
    const start = new Date();
    start.setDate(today.getDate() - diff);
    startInput.value = formatLocal(start);
    endInput.value = formatLocal(today);
  } else if (type === 'month') {
    const start = new Date(today.getFullYear(), today.getMonth(), 1);
    startInput.value = formatLocal(start);
    endInput.value = formatLocal(today);
  } else if (type === 'all') {
    startInput.value = '';
    endInput.value = '';
  }

  loadMntLogsHistory();
}

function applyMntHistoryClientFilters() {
  const taskType = document.getElementById('mntHistTaskType').value;
  const location = document.getElementById('mntHistLocation').value;
  const category = document.getElementById('mntHistCategory').value;
  const performer = document.getElementById('mntHistPerformer').value;
  const groupBy = document.getElementById('mntHistGroupBy').value;
  const search = (document.getElementById('mntHistSearchInput').value || '').trim().toLowerCase();

  // Calculate Active Filters Count
  let activeFilters = 0;
  if (taskType !== 'الكل') activeFilters++;
  if (location !== 'الكل') activeFilters++;
  if (category !== 'الكل') activeFilters++;
  if (performer !== 'الكل') activeFilters++;
  if (search) activeFilters++;
  if (groupBy !== 'none') activeFilters++;

  const badge = document.getElementById('mntHistActiveFilterBadge');
  if (badge) {
    badge.textContent = `${activeFilters} فلاتر نشطة`;
    badge.classList.toggle('hidden', activeFilters === 0);
  }

  const filtered = currentMntHistoryLogs.filter(log => {
    if (taskType !== 'الكل' && log.taskType !== taskType) return false;
    if (location !== 'الكل' && log.location !== location) return false;
    if (category !== 'الكل' && log.category !== category) return false;

    const perfNames = (log.performers && Array.isArray(log.performers))
      ? log.performers.map(p => (p.name || '').trim())
      : [(log.performer || '').trim()];

    if (performer !== 'الكل' && !perfNames.includes(performer)) return false;

    if (search) {
      const idMatch = (log.itemId || '').toLowerCase().includes(search);
      const nameMatch = (log.itemName || '').toLowerCase().includes(search);
      const notesMatch = (log.notes || '').toLowerCase().includes(search);
      const perfMatch = perfNames.join(' ').toLowerCase().includes(search);
      const recMatch = (log.recorder || '').toLowerCase().includes(search);
      if (!idMatch && !nameMatch && !notesMatch && !perfMatch && !recMatch) return false;
    }

    return true;
  });

  document.getElementById('mntHistRecordCount').textContent = filtered.length;
  renderMntHistoryCards(filtered, groupBy);
}

function renderMntHistoryCards(logs, groupBy) {
  const container = document.getElementById('mntHistoryCardsContainer');
  container.innerHTML = '';

  if (!logs || logs.length === 0) {
    container.innerHTML = `
      <div class="py-12 text-center text-slate-400 bg-white rounded-2xl border border-slate-200/80">
        <i class="fas fa-inbox text-4xl mb-3 text-slate-300"></i>
        <p class="font-bold">لا توجد سجلات مطابقة للفلاتر المحددة</p>
      </div>
    `;
    return;
  }

  if (groupBy === 'none') {
    logs.forEach(log => {
      container.appendChild(createMntLogCardElement(log));
    });
  } else {
    // Grouping Logic
    const groups = {};
    logs.forEach(log => {
      let key = log[groupBy] || 'غير محدد';
      if (groupBy === 'date') key = log.date;
      if (!groups[key]) groups[key] = [];
      groups[key].push(log);
    });

    for (const [groupName, groupLogs] of Object.entries(groups)) {
      const groupBlock = document.createElement('div');
      groupBlock.className = 'flex flex-col gap-2.5 mb-3';

      let iconClass = 'fa-folder';
      if (groupBy === 'date') iconClass = 'fa-calendar-alt text-blue-500';
      else if (groupBy === 'location') iconClass = 'fa-map-marker-alt text-rose-500';
      else if (groupBy === 'category') iconClass = 'fa-tags text-purple-500';
      else if (groupBy === 'performer') iconClass = 'fa-user-check text-emerald-500';
      else if (groupBy === 'taskType') iconClass = 'fa-broom text-amber-500';

      groupBlock.innerHTML = `
        <div class="flex justify-between items-center bg-slate-100/90 px-3.5 py-2 rounded-xl border border-slate-200 text-slate-700">
          <span class="font-black text-xs flex items-center gap-2"><i class="fas ${iconClass}"></i> ${groupName}</span>
          <span class="text-[11px] font-bold text-slate-500 bg-white px-2 py-0.5 rounded-md border border-slate-200">${groupLogs.length} عمليات</span>
        </div>
      `;

      const cardsWrap = document.createElement('div');
      cardsWrap.className = 'flex flex-col gap-2.5';
      groupLogs.forEach(log => {
        cardsWrap.appendChild(createMntLogCardElement(log));
      });

      groupBlock.appendChild(cardsWrap);
      container.appendChild(groupBlock);
    }
  }
}

function getDirectDriveImgUrl(url) {
  if (!url) return '';
  const match = String(url).match(/[-\w]{25,}/);
  if (match && match[0]) {
    return 'https:\x2F\x2Flh3.googleusercontent.com/d/' + match[0];
  }
  return url;
}

function createMntLogCardElement(log) {
  const card = document.createElement('div');
  const isClean = log.taskType === 'نظافة';
  
  // High-contrast outer border with colored right-side indicator bar
  const borderClasses = isClean
    ? 'border-2 border-slate-200 border-r-[6px] border-r-blue-600 shadow-[0_2px_8px_-2px_rgba(37,99,235,0.12)] hover:border-blue-300'
    : 'border-2 border-slate-200 border-r-[6px] border-r-amber-500 shadow-[0_2px_8px_-2px_rgba(245,158,11,0.12)] hover:border-amber-300';

  card.className = `bg-white rounded-2xl p-4.5 ${borderClasses} flex flex-col gap-3 transition-all hover:shadow-md`;

  const taskBadge = isClean 
    ? '<span class="bg-blue-50 text-blue-700 px-2.5 py-1 rounded-lg text-xs font-black border border-blue-200 flex items-center gap-1.5"><i class="fas fa-broom"></i> نظافة وتطهير</span>'
    : '<span class="bg-amber-50 text-amber-700 px-2.5 py-1 rounded-lg text-xs font-black border border-amber-200 flex items-center gap-1.5"><i class="fas fa-wrench"></i> صيانة وفحص</span>';

  // Multi-performers formatted badges
  let performersHTML = '';
  if (log.performers && Array.isArray(log.performers)) {
    performersHTML = log.performers.map(p => {
      let ratingBadge = '';
      if (p.rating !== null && p.rating !== undefined) {
        const rColor = p.rating >= 8.5 ? 'bg-emerald-100 text-emerald-800' : (p.rating >= 6.5 ? 'bg-amber-100 text-amber-800' : 'bg-rose-100 text-rose-800');
        ratingBadge = `<span class="px-1.5 py-0.5 rounded font-black text-[10px] ${rColor} mr-1">${p.rating}/10</span>`;
      }
      return `<span class="bg-slate-100 text-slate-800 px-2 py-0.5 rounded-md font-bold text-xs inline-flex items-center">${p.name} ${ratingBadge}</span>`;
    }).join(' ');
  } else {
    performersHTML = `<span class="font-bold text-slate-800">${log.performer || '-'}</span>`;
  }

  // Build Image Thumbnails Tray
  let imagesHTML = '';
  if (log.images && log.images.length > 0) {
    const thumbs = log.images.slice(0, 3).map(url => {
      const directUrl = getDirectDriveImgUrl(url);
      return `<img src="${directUrl}" class="w-12 h-12 object-cover rounded-lg border border-slate-200 shadow-sm">`;
    }).join('');

    const remainingCount = log.images.length - 3;
    const moreBadge = remainingCount > 0 ? `<div class="w-12 h-12 bg-slate-100 rounded-lg border border-slate-200 flex items-center justify-center font-bold text-xs text-slate-500">+${remainingCount}</div>` : '';

    imagesHTML = `
      <div onclick="openMntAlbumByLogId('${log.id}')" class="flex items-center gap-2 mt-1 p-2 bg-slate-50 hover:bg-blue-50/60 rounded-xl border border-slate-200/60 cursor-pointer transition-all">
        <div class="flex items-center gap-1.5">
          ${thumbs}
          ${moreBadge}
        </div>
        <div class="mr-auto text-left">
          <span class="text-xs font-bold text-blue-600 hover:text-blue-700 flex items-center gap-1">
            <i class="fas fa-images"></i> عرض الألبوم (${log.images.length})
          </span>
        </div>
      </div>
    `;
  }

  card.innerHTML = `
    <div class="flex justify-between items-start gap-2">
      <div class="flex items-center gap-2">
        ${taskBadge}
        <span id="mntBadge_${log.id}" class="text-xs font-black text-slate-700 bg-slate-100 hover:bg-amber-100 hover:text-amber-900 px-2 py-0.5 rounded-md border border-slate-200 cursor-pointer select-none transition-colors" title="اضغط مطولاً للتعديل السري">#${log.itemId}</span>
      </div>
      <div class="text-left">
        <span class="text-xs font-bold text-slate-600">${log.date}</span>
        <span class="text-[10px] text-slate-400 block">${log.timestamp ? log.timestamp.substring(11) : ''}</span>
      </div>
    </div>

    <div>
      <h4 class="text-sm font-black text-slate-800 m-0">${log.itemName}</h4>
      <div class="flex flex-wrap gap-1.5 mt-1.5">
        <span class="text-[11px] font-bold text-slate-500 bg-slate-100 px-2 py-0.5 rounded-md"><i class="fas fa-map-marker-alt text-slate-400 ml-1"></i>${log.location}</span>
        <span class="text-[11px] font-bold text-slate-500 bg-slate-100 px-2 py-0.5 rounded-md"><i class="fas fa-tag text-slate-400 ml-1"></i>${log.category}</span>
      </div>
    </div>

    <div class="bg-slate-50/70 p-2.5 rounded-xl border border-slate-100 flex flex-col gap-1 text-xs">
      <div class="flex justify-between items-center flex-wrap gap-1">
        <div class="text-slate-600 font-bold flex items-center flex-wrap gap-1">
          <i class="fas fa-user-check text-blue-500 ml-1"></i> القائمون: ${performersHTML}
        </div>
        <span class="text-[10px] text-slate-400 font-medium">المسجل: ${log.recorder}</span>
      </div>
      ${log.notes ? `<div class="mt-1 pt-1 border-t border-slate-200/60 text-slate-600 text-xs font-medium"><i class="fas fa-comment-alt text-slate-400 ml-1"></i>${log.notes}</div>` : ''}
    </div>

    ${imagesHTML}
  `;

  // Long-press / Long-click event binding for secret edit trigger
  setTimeout(() => {
    const badgeEl = card.querySelector(`#mntBadge_${log.id}`);
    if (badgeEl) {
      let pressTimer;
      const startPress = () => {
        pressTimer = setTimeout(() => {
          promptMntSecretEdit(log.id);
        }, 750); // 750ms threshold for long tap
      };
      const cancelPress = () => clearTimeout(pressTimer);

      badgeEl.addEventListener('touchstart', startPress, { passive: true });
      badgeEl.addEventListener('touchend', cancelPress);
      badgeEl.addEventListener('touchcancel', cancelPress);
      badgeEl.addEventListener('mousedown', startPress);
      badgeEl.addEventListener('mouseup', cancelPress);
      badgeEl.addEventListener('mouseleave', cancelPress);
    }
  }, 0);

  return card;
}

function setMntRatingsQuickDate(type) {
  const today = new Date();
  const startInput = document.getElementById('mntRatingsStartDate');
  const endInput = document.getElementById('mntRatingsEndDate');
  const formatLocal = (d) => `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;

  if (type === 'today') {
    startInput.value = formatLocal(today);
    endInput.value = formatLocal(today);
  } else if (type === 'week') {
    const day = today.getDay();
    const diff = day === 6 ? 0 : day + 1;
    const start = new Date();
    start.setDate(today.getDate() - diff);
    startInput.value = formatLocal(start);
    endInput.value = formatLocal(today);
  } else if (type === 'month') {
    const start = new Date(today.getFullYear(), today.getMonth(), 1);
    startInput.value = formatLocal(start);
    endInput.value = formatLocal(today);
  } else if (type === 'all') {
    startInput.value = '';
    endInput.value = '';
  }

  loadMntRatingsReport();
}

function loadMntRatingsReport() {
  const startDate = document.getElementById('mntRatingsStartDate') ? document.getElementById('mntRatingsStartDate').value : '';
  const endDate = document.getElementById('mntRatingsEndDate') ? document.getElementById('mntRatingsEndDate').value : '';

  const leaderboard = document.getElementById('mntStaffLeaderboard');
  const taskLogs = document.getElementById('mntRatingsTaskLogs');

  leaderboard.innerHTML = '<div class="col-span-full py-8 text-center text-slate-400"><i class="fas fa-spinner fa-spin text-2xl mb-2"></i><br>جاري احتساب التقييمات...</div>';
  taskLogs.innerHTML = '';

  google.script.run
    .withSuccessHandler(function(logs) {
      renderCleanlinessRatings(logs || []);
    })
    .withFailureHandler(function(err) {
      leaderboard.innerHTML = `<div class="col-span-full py-6 text-center text-rose-500 font-bold">${err.message}</div>`;
    })
    .getMntLogsHistory({ startDate: startDate, endDate: endDate, taskType: 'نظافة', location: 'الكل', category: 'الكل' });
}

function renderCleanlinessRatings(logs) {
  const staffStats = {};
  let totalRatingSum = 0;
  let totalRatedCount = 0;

  logs.forEach(log => {
    if (log.performers && Array.isArray(log.performers)) {
      log.performers.forEach(p => {
        const name = (p.name || '').trim();
        if (!name) return;
        if (!staffStats[name]) {
          staffStats[name] = { name: name, totalTasks: 0, ratedTasks: 0, sumRating: 0, logs: [] };
        }
        staffStats[name].totalTasks++;
        staffStats[name].logs.push(log);

        if (p.rating !== null && p.rating !== undefined && !isNaN(p.rating)) {
          staffStats[name].ratedTasks++;
          staffStats[name].sumRating += parseFloat(p.rating);
          totalRatingSum += parseFloat(p.rating);
          totalRatedCount++;
        }
      });
    }
  });

  // Calculate Overall KPIs
  const overallAvg = totalRatedCount > 0 ? (totalRatingSum / totalRatedCount).toFixed(1) : '0.0';
  document.getElementById('mntKpiAvgRating').textContent = overallAvg;
  document.getElementById('mntKpiTotalCleanOps').textContent = logs.length;

  const staffArray = Object.values(staffStats).map(s => {
    const avg = s.ratedTasks > 0 ? parseFloat((s.sumRating / s.ratedTasks).toFixed(1)) : 0;
    return { ...s, avgRating: avg };
  });

  staffArray.sort((a, b) => b.avgRating - a.avgRating || b.totalTasks - a.totalTasks);

  document.getElementById('mntKpiTopPerformer').textContent = staffArray.length > 0 && staffArray[0].avgRating > 0 ? `${staffArray[0].name} (${staffArray[0].avgRating}/10)` : '-';

  // Render Leaderboard Cards
  const leaderboard = document.getElementById('mntStaffLeaderboard');
  leaderboard.innerHTML = '';

  if (staffArray.length === 0) {
    leaderboard.innerHTML = '<div class="col-span-full py-8 text-center text-slate-400 font-bold bg-white rounded-2xl border border-slate-200">لا توجد عمليات نظافة مسجلة لهذه الفترة</div>';
    return;
  }

  staffArray.forEach((staff, rank) => {
    const card = document.createElement('div');
    const rColor = staff.avgRating >= 8.5 ? 'text-emerald-600 border-emerald-300 bg-emerald-50' : (staff.avgRating >= 6.5 ? 'text-amber-600 border-amber-300 bg-amber-50' : 'text-rose-600 border-rose-300 bg-rose-50');
    const barWidth = `${(staff.avgRating / 10) * 100}%`;

    card.className = 'bg-white rounded-2xl p-4 border-2 border-slate-200/90 shadow-sm flex flex-col justify-between hover:shadow-md transition-all';
    card.innerHTML = `
      <div>
        <div class="flex justify-between items-start mb-2">
          <div class="flex items-center gap-2">
            <span class="w-6 h-6 rounded-full bg-slate-100 text-slate-700 font-black text-xs flex items-center justify-center border border-slate-200">#${rank + 1}</span>
            <h4 class="text-sm font-black text-slate-800 m-0">${staff.name}</h4>
          </div>
          <span class="text-xs font-black px-2.5 py-1 rounded-xl border ${rColor}">${staff.avgRating} / 10</span>
        </div>

        <div class="w-full bg-slate-100 h-2 rounded-full overflow-hidden my-2.5">
          <div class="h-full rounded-full transition-all duration-500 ${staff.avgRating >= 8.5 ? 'bg-emerald-500' : (staff.avgRating >= 6.5 ? 'bg-amber-500' : 'bg-rose-500')}" style="width: ${barWidth};"></div>
        </div>

        <div class="flex justify-between text-xs text-slate-500 font-bold">
          <span>المهام المنجزة: <strong class="text-slate-700">${staff.totalTasks}</strong></span>
          <span>المهام المقيمة: <strong class="text-slate-700">${staff.ratedTasks}</strong></span>
        </div>
      </div>
    `;
    leaderboard.appendChild(card);
  });

  // Render Cleaning Task Logs
  const taskLogs = document.getElementById('mntRatingsTaskLogs');
  taskLogs.innerHTML = '';
  logs.forEach(log => {
    taskLogs.appendChild(createMntLogCardElement(log));
  });
}

function exportMntHistoryPDF() {
  if (!currentMntHistoryLogs || currentMntHistoryLogs.length === 0) {
    return alert('لا توجد سجلات متاحة للتصدير');
  }

  const startDate = document.getElementById('mntHistStartDate') ? document.getElementById('mntHistStartDate').value : '';
  const endDate = document.getElementById('mntHistEndDate') ? document.getElementById('mntHistEndDate').value : '';
  const taskType = document.getElementById('mntHistTaskType') ? document.getElementById('mntHistTaskType').value : 'الكل';
  const location = document.getElementById('mntHistLocation') ? document.getElementById('mntHistLocation').value : 'الكل';
  const category = document.getElementById('mntHistCategory') ? document.getElementById('mntHistCategory').value : 'الكل';
  const performer = document.getElementById('mntHistPerformer') ? document.getElementById('mntHistPerformer').value : 'الكل';
  const groupBy = document.getElementById('mntHistGroupBy') ? document.getElementById('mntHistGroupBy').value : 'none';
  const search = document.getElementById('mntHistSearchInput') ? document.getElementById('mntHistSearchInput').value.trim() : '';

  const btn = document.getElementById('btnExportMntPDF');
  const originalHtml = btn.innerHTML;
  btn.disabled = true;
  btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> جاري إنشاء الـ PDF...';

  showLoading();

  google.script.run
    .withSuccessHandler(function(res) {
      hideLoading();
      btn.disabled = false;
      btn.innerHTML = originalHtml;
      if (res.url) {
        window.open(res.url, '_blank');
      }
    })
    .withFailureHandler(function(err) {
      hideLoading();
      btn.disabled = false;
      btn.innerHTML = originalHtml;
      alert('خطأ في تصدير الـ PDF: ' + err.message);
    })
    .exportMntLogsPDF({
      startDate: startDate,
      endDate: endDate,
      taskType: taskType,
      location: location,
      category: category,
      performer: performer,
      groupBy: groupBy,
      search: search,
      exportedBy: currentUser || 'مسؤول النظام'
    });
}

var activeEditLogId = null;
var editRetainedImages = [];
var editUploadedNewImages = [];
var activeEditLogObj = null;

var activeVerifiedPasscode = '';

function promptMntSecretEdit(logId) {
  activeEditLogId = logId;
  activeVerifiedPasscode = '';
  document.getElementById('mntSecretPinInput').value = '';
  
  const subEl = document.getElementById('mntPasscodeSubtitle');
  if (subEl) {
    subEl.textContent = `أدخل رمز المرور الخاص بحساب (${currentUser}) لتعديل السجل`;
  }

  document.getElementById('mntPasscodeModal').style.display = 'flex';
  setTimeout(() => document.getElementById('mntSecretPinInput').focus(), 100);
}

function closeMntPasscodeModal() {
  document.getElementById('mntPasscodeModal').style.display = 'none';
}

function submitMntPasscode(e) {
  e.preventDefault();
  const enteredPin = document.getElementById('mntSecretPinInput').value.trim();
  if (!enteredPin) return alert('الرجاء إدخال رمز المرور');

  const btn = e.target.querySelector('button[type="submit"]');
  const originalHtml = btn.innerHTML;
  btn.disabled = true;
  btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i>';

  google.script.run
    .withSuccessHandler(function(isValid) {
      btn.disabled = false;
      btn.innerHTML = originalHtml;

      if (isValid) {
        activeVerifiedPasscode = enteredPin;
        closeMntPasscodeModal();
        openMntEditModal(activeEditLogId);
      } else {
        alert(`رمز المرور غير صحيح لحساب "${currentUser}"`);
        document.getElementById('mntSecretPinInput').value = '';
      }
    })
    .withFailureHandler(function(err) {
      btn.disabled = false;
      btn.innerHTML = originalHtml;
      alert('خطأ في التحقق من المستخدم: ' + err.message);
    })
    .verifyUserPasscode(currentUser, enteredPin);
}

function openMntEditModal(logId) {
  const log = currentMntHistoryLogs.find(l => l.id === logId);
  if (!log) return;
  activeEditLogObj = log;

  document.getElementById('mntEditModalTitle').innerHTML = `<i class="fas fa-edit text-amber-400"></i> تعديل سجل: #${log.itemId} - ${log.itemName}`;
  document.getElementById('mntEditModalSubtitle').textContent = `${log.taskType} • ${log.location} • ${log.category}`;
  document.getElementById('mntEditDate').value = log.date;
  document.getElementById('mntEditNotes').value = log.notes || '';

  // Populate Performers
  const listContainer = document.getElementById('mntEditPerformersList');
  listContainer.innerHTML = '';
  if (log.performers && Array.isArray(log.performers) && log.performers.length > 0) {
    log.performers.forEach(p => addMntEditPerformerRow(p.name, p.rating));
  } else {
    addMntEditPerformerRow(log.performer || '', null);
  }

  // Populate Existing Images
  editRetainedImages = [...(log.images || [])];
  editUploadedNewImages = [];
  renderMntEditExistingImages();
  renderMntEditNewImages();

  document.getElementById('mntEditRecordModal').style.display = 'flex';
}

function closeMntEditModal() {
  document.getElementById('mntEditRecordModal').style.display = 'none';
  activeEditLogId = null;
  activeEditLogObj = null;
  editRetainedImages = [];
  editUploadedNewImages = [];
}

function removeMntExistingImage(idx) {
  editRetainedImages.splice(idx, 1);

}

function handleMntEditMultiFiles(e) {
  const files = Array.from(e.target.files);
  if (!files || files.length === 0) return;

  showLoading();
  let loaded = 0;

  files.forEach(file => {
    compressImage(file, function(compressedData) {
      editUploadedNewImages.push(compressedData);
      loaded++;
      if (loaded === files.length) {
        hideLoading();
        renderMntEditNewImages();
      }
    });
  });

  e.target.value = '';
}

function renderMntEditNewImages() {
  const tray = document.getElementById('mntEditNewImagesTray');
  tray.innerHTML = '';

  editUploadedNewImages.forEach((imgBase64, idx) => {
    const card = document.createElement('div');
    card.className = 'relative aspect-square rounded-xl overflow-hidden border border-slate-200 shadow-sm';
    card.innerHTML = `
      <img src="${imgBase64}" class="w-full h-full object-cover">
      <button type="button" onclick="removeMntEditNewImage(${idx})" class="absolute top-1 right-1 w-6 h-6 rounded-full bg-rose-600 text-white flex items-center justify-center text-xs shadow-md transition-transform hover:scale-110">
        <i class="fas fa-times"></i>
      </button>
    `;
    tray.appendChild(card);
  });
}

function removeMntEditNewImage(idx) {
  editUploadedNewImages.splice(idx, 1);
  renderMntEditNewImages();
}

function saveMntEditRecord(e) {
  e.preventDefault();
  if (!activeEditLogObj) return;

  const date = document.getElementById('mntEditDate').value;
  const notes = document.getElementById('mntEditNotes').value.trim();

  // Validate Performers
  const rows = document.querySelectorAll('.mnt-edit-perf-row');
  const performers = [];
  let missingRatingPerformer = null;

  rows.forEach(row => {
    const name = (row.querySelector('.mnt-edit-perf-name').value || '').trim();
    const ratingInput = row.querySelector('.mnt-edit-perf-rating');
    let rating = null;

    if (!name) return;

    if (activeEditLogObj.taskType === 'نظافة' && ratingInput) {
      const parsedRating = parseInt(convertArabicToEnglishNums(ratingInput.value));
      if (isNaN(parsedRating) || parsedRating < 1 || parsedRating > 10) {
        missingRatingPerformer = name;
      } else {
        rating = parsedRating;
      }
    }

    performers.push({ name: name, rating: rating });
  });

  if (performers.length === 0) {
    return alert('يجب تحديد قائم بالعمل واحد على الأقل');
  }

  if (activeEditLogObj.taskType === 'نظافة' && missingRatingPerformer !== null) {
    return alert(`الرجاء إدخال تقييم النظافة (1-10) لـ: "${missingRatingPerformer}"`);
  }

  const payload = {
    user: currentUser,
    passcode: activeVerifiedPasscode,
    logId: activeEditLogObj.id,
    itemId: activeEditLogObj.itemId,
    date: date,
    performers: performers,
    notes: notes,
    existingImages: editRetainedImages,
    newImages: editUploadedNewImages
  };

  const btn = document.getElementById('btnSaveMntEdit');
  btn.disabled = true;
  btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> جاري حفظ التعديلات...';
  showLoading();

  google.script.run
    .withSuccessHandler(function(res) {
      hideLoading();
      btn.disabled = false;
      btn.innerHTML = '<i class="fas fa-save"></i> حفظ التعديلات';
      alert(res.message);
      closeMntEditModal();
      loadMntLogsHistory(); // Live reload history view
    })
    .withFailureHandler(function(err) {
      hideLoading();
      btn.disabled = false;
      btn.innerHTML = '<i class="fas fa-save"></i> حفظ التعديلات';
      alert('خطأ أثناء التعديل: ' + err.message);
    })
    .updateMntLog(payload);
}
</script>
